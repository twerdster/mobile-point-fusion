cmake_minimum_required(VERSION 3.20)
project(points_fusion_offline LANGUAGES CXX)
option(INTEGER_FLOAT_STORAGE "Use RGBA32UI float-bit storage (the iOS mode)" OFF)
if(INTEGER_FLOAT_STORAGE)
  add_compile_definitions(PF_INTEGER_FLOAT_STORAGE=1)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SANITIZERS "Enable address and undefined-behaviour diagnostics" OFF)
set(MESA_EGL_VENDOR_FILE "/usr/share/glvnd/egl_vendor.d/50_mesa.json" CACHE FILEPATH "Mesa EGL vendor JSON for headless CTest")
find_package(PkgConfig REQUIRED)
pkg_check_modules(EGL REQUIRED IMPORTED_TARGET egl)
pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)
add_executable(points_fusion src/main.cpp)
target_link_libraries(points_fusion PRIVATE PkgConfig::EGL PkgConfig::GLES)
target_compile_definitions(points_fusion PRIVATE SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_options(points_fusion PRIVATE -Wall -Wextra -Wpedantic)
add_executable(original_faults tests/original_faults.cpp)
if(SANITIZERS)
  foreach(target points_fusion original_faults)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME cpu_regressions COMMAND points_fusion --cpu-test)
add_test(NAME gles_shader_references COMMAND points_fusion --gpu-test)
set_tests_properties(gles_shader_references PROPERTIES ENVIRONMENT "LIBGL_ALWAYS_SOFTWARE=1;EGL_PLATFORM=surfaceless;__EGL_VENDOR_LIBRARY_FILENAMES=${MESA_EGL_VENDOR_FILE};LP_NUM_THREADS=2")

option(NATIVE_UI "Build the X11/EGL native interactive frontend" OFF)
if(NATIVE_UI)
  pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11)
  add_executable(points_fusion_ui platforms/linux/main.cpp)
  target_link_libraries(points_fusion_ui PRIVATE PkgConfig::EGL PkgConfig::GLES PkgConfig::X11)
  target_compile_definitions(points_fusion_ui PRIVATE SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
  target_compile_options(points_fusion_ui PRIVATE -Wall -Wextra -Wpedantic)
  if(SANITIZERS)
    target_compile_options(points_fusion_ui PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
    target_link_options(points_fusion_ui PRIVATE -fsanitize=address,undefined)
  endif()
endif()
